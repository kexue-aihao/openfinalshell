cmake_minimum_required(VERSION 3.16)
project(ofs_rdp_worker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# FreeRDP is enabled by default. Development builds may explicitly disable it,
# while release/CI builds pass OFS_RDP_REQUIRE_FREERDP=ON to fail closed when
# the native SDK is unavailable.
option(OFS_RDP_ENABLE_FREERDP "Enable the FreeRDP backend when available" ON)
option(OFS_RDP_REQUIRE_FREERDP "Fail configure when the FreeRDP backend cannot be enabled" OFF)

set(OFS_RDP_HAS_FREERDP 0)
set(OFS_RDP_FREERDP_MAJOR 0)

add_executable(ofs-rdp-worker main.cpp freerdp_adapter.cpp unicode.cpp file_clipboard.cpp rdp_clipboard_core.cpp audio_backend.cpp)
find_package(Threads REQUIRED)
target_link_libraries(ofs-rdp-worker PRIVATE Threads::Threads)
target_link_libraries(ofs-rdp-worker PRIVATE ${CMAKE_DL_LIBS})
if (APPLE)
  set_target_properties(ofs-rdp-worker PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
elseif (UNIX)
  set_target_properties(ofs-rdp-worker PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
if (WIN32)
  target_link_libraries(ofs-rdp-worker PRIVATE ws2_32 ole32 shell32 uuid)
  # The native adapter probes the default playback device before enabling
  # FreeRDP rdpsnd, and the bundled FreeRDP backend uses WinMM waveOut.
  target_link_libraries(ofs-rdp-worker PRIVATE winmm)
endif()
if (MSVC)
  target_compile_definitions(ofs-rdp-worker PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()

if (OFS_RDP_ENABLE_FREERDP)
  find_package(PkgConfig QUIET)
  if (PkgConfig_FOUND)
    pkg_check_modules(FREERDP2 QUIET freerdp2 freerdp-client2)
    pkg_check_modules(FREERDP3 QUIET freerdp3 freerdp-client3)
    pkg_check_modules(WINPR2 QUIET winpr2)
    pkg_check_modules(WINPR3 QUIET winpr3)
    pkg_check_modules(FREERDP_GDI3 QUIET freerdp3-gdi)
  endif()
  if (FREERDP2_FOUND OR FREERDP3_FOUND)
    # Override the default above only when the SDK and linker metadata are
    # both present. The adapter uses libfreerdp-client symbols for cliprdr and
    # channel loading, so the core and client packages must be found together.
    set(OFS_RDP_HAS_FREERDP 1)
    if (FREERDP3_FOUND)
      set(OFS_RDP_FREERDP_MAJOR 3)
      set(FREERDP_INCLUDE_DIRS ${FREERDP3_INCLUDE_DIRS})
      set(FREERDP_LIBRARY_DIRS ${FREERDP3_LIBRARY_DIRS})
      set(FREERDP_LIBRARIES ${FREERDP3_LIBRARIES})
      set(FREERDP_CFLAGS_OTHER ${FREERDP3_CFLAGS_OTHER})
      set(FREERDP_LDFLAGS_OTHER ${FREERDP3_LDFLAGS_OTHER})
      list(APPEND FREERDP_LIBRARIES ${WINPR3_LIBRARIES})
      list(APPEND FREERDP_LIBRARY_DIRS ${WINPR3_LIBRARY_DIRS})
      list(APPEND FREERDP_LIBRARIES ${FREERDP_GDI3_LIBRARIES})
      list(APPEND FREERDP_LIBRARY_DIRS ${FREERDP_GDI3_LIBRARY_DIRS})
    else()
      set(OFS_RDP_FREERDP_MAJOR 2)
      set(FREERDP_INCLUDE_DIRS ${FREERDP2_INCLUDE_DIRS})
      set(FREERDP_LIBRARY_DIRS ${FREERDP2_LIBRARY_DIRS})
      set(FREERDP_LIBRARIES ${FREERDP2_LIBRARIES})
      set(FREERDP_CFLAGS_OTHER ${FREERDP2_CFLAGS_OTHER})
      set(FREERDP_LDFLAGS_OTHER ${FREERDP2_LDFLAGS_OTHER})
      list(APPEND FREERDP_LIBRARIES ${WINPR2_LIBRARIES})
      list(APPEND FREERDP_LIBRARY_DIRS ${WINPR2_LIBRARY_DIRS})
    endif()
    target_include_directories(ofs-rdp-worker PRIVATE ${FREERDP_INCLUDE_DIRS})
    target_link_directories(ofs-rdp-worker PRIVATE ${FREERDP_LIBRARY_DIRS})
    target_link_libraries(ofs-rdp-worker PRIVATE ${FREERDP_LIBRARIES})
    target_compile_options(ofs-rdp-worker PRIVATE ${FREERDP_CFLAGS_OTHER})
    target_link_options(ofs-rdp-worker PRIVATE ${FREERDP_LDFLAGS_OTHER})
  else()
    find_package(FreeRDP3 CONFIG QUIET)
    find_package(FreeRDP-Client3 CONFIG QUIET)
    find_package(WinPR3 CONFIG QUIET)
    find_package(FreeRDP CONFIG QUIET)
    find_package(FreeRDP-Client CONFIG QUIET)
    find_package(WinPR CONFIG QUIET)
    if (TARGET freerdp3 AND TARGET freerdp-client3 AND TARGET winpr3)
      set(OFS_RDP_FREERDP_MAJOR 3)
      set(OFS_RDP_HAS_FREERDP 1)
      target_link_libraries(ofs-rdp-worker PRIVATE freerdp3 freerdp-client3 winpr3)
    elseif (TARGET freerdp AND TARGET freerdp-client AND TARGET winpr)
      set(OFS_RDP_FREERDP_MAJOR 2)
      set(OFS_RDP_HAS_FREERDP 1)
      target_link_libraries(ofs-rdp-worker PRIVATE freerdp freerdp-client winpr)
    endif()
  endif()

  if (NOT OFS_RDP_HAS_FREERDP)
    if (OFS_RDP_REQUIRE_FREERDP)
      message(FATAL_ERROR "OFS_RDP_REQUIRE_FREERDP is ON but FreeRDP core+client 2.x/3.x pkg-config modules or CMake targets were not found")
    endif()
    message(WARNING "OFS_RDP_ENABLE_FREERDP is ON but FreeRDP core+client 2.x/3.x was not found; building protocol-only worker")
  endif()
endif()

target_compile_definitions(ofs-rdp-worker PRIVATE OFS_RDP_HAS_FREERDP=${OFS_RDP_HAS_FREERDP})
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/rdp-worker-sdk.json"
  CONTENT "{\"freerdpMajor\":${OFS_RDP_FREERDP_MAJOR}}\n")

if (APPLE)
  enable_language(OBJCXX)
  set_property(TARGET ofs-rdp-worker PROPERTY OBJCXX_STANDARD 17)
  set_property(TARGET ofs-rdp-worker PROPERTY OBJCXX_STANDARD_REQUIRED ON)
  target_sources(ofs-rdp-worker PRIVATE clipboard_macos.mm file_clipboard_unix.cpp)
  set_source_files_properties(clipboard_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(ofs-rdp-worker PRIVATE "-framework AppKit")
elseif (UNIX)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  target_sources(ofs-rdp-worker PRIVATE clipboard_gtk.cpp file_clipboard_unix.cpp)
  target_link_libraries(ofs-rdp-worker PRIVATE PkgConfig::GTK3)
endif()

    target_compile_options(ofs-rdp-audio-backend-test PRIVATE $<TARGET_PROPERTY:ofs-rdp-worker,COMPILE_OPTIONS>)
    target_link_options(ofs-rdp-audio-backend-test PRIVATE $<TARGET_PROPERTY:ofs-rdp-worker,LINK_OPTIONS>)
    if (MSVC)
      target_compile_options(ofs-rdp-audio-backend-test PRIVATE /UNDEBUG)

      target_compile_options(ofs-rdp-audio-backend-test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME worker_audio_backend COMMAND ofs-rdp-audio-backend-test)
  endif()
  if (WIN32)
    # file_clipboard.cpp now decodes/ad encodes UTF-8 via unicode.cpp for the
    # local-clipboard text helpers, so the test target must link it too.
    add_executable(ofs-rdp-file-clipboard-test test/file_clipboard.test.cpp unicode.cpp)
    target_link_libraries(ofs-rdp-file-clipboard-test PRIVATE ole32 shell32 uuid)
    target_include_directories(ofs-rdp-file-clipboard-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
